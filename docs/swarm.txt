Pattern 1: Ephemeral Sub-Agents (Dynamic Factory)

Here, a Lead Architect agent starts with zero pre-built specialists. Instead, it has a spawn_sub_agent tool that instantiates a brand-new, single-purpose agent with a custom persona on the fly, runs it on a sub-task, and discards it when finished.
python
import json
from openai import OpenAI

client = OpenAI(base_url="http://localhost:1234/v1", api_key="lm-studio")
MODEL = "local-model"  # Uses whatever is loaded in LM Studio

# Guardrail: Prevent infinite spawning
MAX_SPAWNS = 4
spawn_count = 0


def run_ephemeral_worker(role: str, system_prompt: str, task: str) -> str:
    """Instantiates and runs a disposable specialist agent."""
    print(f"\n  [SPAWNED WORKER: {role}]")
    print(f"  System Prompt: {system_prompt}")
    res = client.chat.completions.create(
        model=MODEL,
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": task},
        ],
        temperature=0.4,
    )
    output = res.choices[0].message.content
    print(f"  [{role} Output]: {output[:120]}...\n")
    return output


TOOLS = [
    {
        "type": "function",
        "function": {
            "name": "spawn_sub_agent",
            "description": "Create a temporary specialist agent with a tailored system prompt to solve a sub-problem.",
            "parameters": {
                "type": "object",
                "properties": {
                    "role": {"type": "string", "description": "Short title, e.g. 'SecurityAuditor'"},
                    "system_prompt": {"type": "string", "description": "Detailed persona and instructions for this worker"},
                    "task": {"type": "string", "description": "Specific sub-task and context for this worker"},
                },
                "required": ["role", "system_prompt", "task"],
            },
        },
    }
]


def run_architect(user_goal: str):
    global spawn_count
    messages = [
        {
            "role": "system",
            "content": (
                "You are a Lead Architect. Do not solve complex problems alone. "
                "Use `spawn_sub_agent` to dynamically create 2-3 specialized experts "
                "(e.g., an adversary, an economist, a systems engineer) to analyze different "
                "angles, then synthesize their findings into a final answer."
            ),
        },
        {"role": "user", "content": user_goal},
    ]

    while True:
        res = client.chat.completions.create(model=MODEL, messages=messages, tools=TOOLS)
        msg = res.choices[0].message
        messages.append(msg)

        if not msg.tool_calls:
            return msg.content

        for call in msg.tool_calls:
            if spawn_count >= MAX_SPAWNS:
                result = "ERROR: Max sub-agent spawn budget reached. Synthesize with what you have."
            else:
                spawn_count += 1
                args = json.loads(call.function.arguments)
                result = run_ephemeral_worker(args["role"], args["system_prompt"], args["task"])

            messages.append({"role": "tool", "tool_call_id": call.id, "content": result})


if __name__ == "__main__":
    answer = run_architect("Design a rate-limiting strategy for a public LLM API gateway.")
    print("\n=== FINAL SYNTHESIS ===\n", answer)

Pattern 2: Self-Modifying Peer Roster (Mutable Graph)

Here, agents live inside a shared roster dictionary (agent_name -> system_prompt). During each turn, the active agent can either hand off to a peer, mutate a peer's system prompt (to fix blind spots or sharpen behavior), or recruit a brand-new peer into the roster.
python
import json
from openai import OpenAI

client = OpenAI(base_url="http://localhost:1234/v1", api_key="lm-studio")
MODEL = "local-model"

# Mutable shared state: Agents can read, edit, or expand this dictionary at runtime
roster = {
    "planner": "You propose concrete solutions. Hand off to 'critic' to stress-test your idea, or mutate peers if their prompts are too weak.",
    "critic": "You find edge cases and flaws. If a new type of expert is needed, recruit one via `upsert_peer`, or hand back to 'planner'.",
}

ROSTER_TOOLS = [
    {
        "type": "function",
        "function": {
            "name": "upsert_peer",
            "description": "Create a new peer agent OR rewrite an existing peer's system prompt to improve swarm performance.",
            "parameters": {
                "type": "object",
                "properties": {
                    "peer_name": {"type": "string", "description": "Lowercase identifier, e.g. 'critic' or 'legal_expert'"},
                    "new_system_prompt": {"type": "string", "description": "The new or updated system instructions for this peer"},
                },
                "required": ["peer_name", "new_system_prompt"],
            },
        },
    },
    {
        "type": "function",
        "function": {
            "name": "handoff",
            "description": "Pass control and a message to another agent currently in the roster.",
            "parameters": {
                "type": "object",
                "properties": {
                    "target_peer": {"type": "string", "description": "Name of the agent in the roster to activate next"},
                    "message": {"type": "string", "description": "Context, draft, or instructions to pass along"},
                },
                "required": ["target_peer", "message"],
            },
        },
    },
]


def run_mutable_swarm(initial_prompt: str, max_turns: int = 6):
    active_agent = "planner"
    shared_transcript = [f"USER GOAL: {initial_prompt}"]

    for turn in range(1, max_turns + 1):
        print(f"\n--- TURN {turn} | ACTIVE: {active_agent} | ROSTER: {list(roster.keys())} ---")

        # Inject the live roster so the active agent knows who exists and how they are configured
        roster_snapshot = "\n".join(f"- {k}: {v}" for k, v in roster.items())
        system_msg = (
            f"You are '{active_agent}'.\n"
            f"YOUR ROLE: {roster[active_agent]}\n\n"
            f"CURRENT LIVE ROSTER:\n{roster_snapshot}\n\n"
            "You may call `upsert_peer` to mutate a peer's prompt or add a new specialist, "
            "then call `handoff` to pass control. If the solution is complete, reply directly without tools."
        )

        messages = [
            {"role": "system", "content": system_msg},
            {"role": "user", "content": "\n\n".join(shared_transcript)},
        ]

        res = client.chat.completions.create(model=MODEL, messages=messages, tools=ROSTER_TOOLS)
        msg = res.choices[0].message

        if not msg.tool_calls:
            print(f"[{active_agent} CONCLUDED]")
            return msg.content

        handed_off = False
        for call in msg.tool_calls:
            args = json.loads(call.function.arguments)

            if call.function.name == "upsert_peer":
                peer = args["peer_name"].lower().strip()
                roster[peer] = args["new_system_prompt"]
                print(f"  [MUTATION] '{active_agent}' updated '{peer}' -> {roster[peer][:90]}...")
                shared_transcript.append(f"[SYSTEM: {active_agent} updated '{peer}' prompt]")

            elif call.function.name == "handoff":
                target = args["target_peer"].lower().strip()
                if target not in roster:
                    shared_transcript.append(f"[ERROR: Peer '{target}' does not exist yet. Create them with upsert_peer first.]")
                else:
                    print(f"  [HANDOFF] {active_agent} -> {target}: {args['message'][:90]}...")
                    shared_transcript.append(f"{active_agent} (to {target}): {args['message']}")
                    active_agent = target
                    handed_off = True

        # If an agent mutated a peer without handing off, keep the turn on the current agent
        if not handed_off:
            continue

    return "Max turns reached.\n\nLast transcript:\n" + "\n".join(shared_transcript[-2:])


if __name__ == "__main__":
    final = run_mutable_swarm("Propose a zero-downtime database migration plan for a 500GB Postgres table.")
    print("\n=== FINAL OUTPUT ===\n", final)

So in both cases we are looping over the agents?

Yes—at the lowest level, almost every multi-agent system is just a while or for loop passing messages and tool results between LLM calls.
The key difference between the two patterns is who controls the loop and how state travels across iterations:
• Pattern 1 (Ephemeral Sub-Agents) is a Nested Loop: The outer while True loop belongs strictly to the Lead Architect. When it spawns a sub-agent, it makes a single blocking API call inside a tool function, gets the result, and always returns control back to the Architect. The sub-agents never talk to each other.
• Pattern 2 (Mutable Peer Roster) is a State-Machine Loop: There is a single flat for turn loop, and a pointer variable (active_agent) tracks who currently holds the microphone. On each iteration, whichever agent is active decides who gets active_agent next—meaning control hops laterally across peers without returning to a central boss.

Why Frameworks Feel Different (Even Though They Do the Exact Same Thing)

Whether you use Strands, LangGraph, or OpenAI's SDK, they are all wrapping this exact same loop under the hood. What frameworks actually add on top of your raw Python loop are:
• Async / Parallel Execution: Running 3 spawned sub-agents simultaneously with asyncio.gather() instead of waiting for them one by one.
• Checkpointing: Serializing messages and roster to SQLite/Postgres after every iteration so you can pause, inspect, or resume a crashed run.
• Streaming: Yielding tokens to a UI in real time while the loop executes.
If you are experimenting locally on a single Mac GPU in LM Studio, your local server processes requests sequentially anyway—so a clean, synchronous Python loop gives you complete transparency with zero framework overhead.

can i save jsonl logging to a postgres table and keep a ledger of all transactions?

Yes—saving your swarm's execution events directly into an append-only PostgreSQL ledger table using JSONB is actually much better than flat JSONL files.
While JSONL is great for quick local tailing, Postgres gives you ACID guarantees, concurrent writes, indexing, and the ability to run SQL queries to compare how prompt mutations impacted token usage, handoff paths, and success rates across runs.

1. Design an Append-Only Event Ledger Schema

Instead of overwriting a single state row, treat your table like an immutable accounting ledger where every agent turn, prompt mutation, and handoff inserts a new row tied to a run_id:
sql
CREATE TABLE swarm_ledger (
    event_id      BIGSERIAL PRIMARY KEY,
    run_id        UUID NOT NULL,
    turn_number   INT NOT NULL,
    event_type    TEXT NOT NULL,        -- e.g., 'turn_start', 'roster_mutation', 'handoff', 'final_output'
    active_agent  TEXT NOT NULL,
    target_agent  TEXT,
    payload       JSONB NOT NULL,       -- Full prompt diff, tool args, or message content
    roster_state  JSONB NOT NULL,       -- Snapshot of all agent prompts at this exact turn
    created_at    TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Index for fast run replays and JSONB querying
CREATE INDEX idx_swarm_ledger_run ON swarm_ledger (run_id, event_id);
CREATE INDEX idx_swarm_ledger_payload ON swarm_ledger USING GIN (payload);

2. Drop-In Python Ledger Helper (psycopg)

Install the modern Postgres driver (pip install "psycopg[binary]") and add a lightweight record_event() function to your swarm loop:
python
import json
import uuid
import psycopg

DB_DSN = "postgresql://postgres:password@localhost:5432/swarm_experiments"

def record_event(
    conn,
    run_id: uuid.UUID,
    turn: int,
    event_type: str,
    active_agent: str,
    payload: dict,
    roster_state: dict,
    target_agent: str | None = None,
):
    with conn.cursor() as cur:
        cur.execute(
            """
            INSERT INTO swarm_ledger 
                (run_id, turn_number, event_type, active_agent, target_agent, payload, roster_state)
            VALUES (%s, %s, %s, %s, %s, %s::jsonb, %s::jsonb)
            """,
            (
                str(run_id),
                turn,
                event_type,
                active_agent,
                target_agent,
                json.dumps(payload),
                json.dumps(roster_state),
            ),
        )
    conn.commit()
Use code with caution.

3. Why This Pays Off for Mutable Swarms

• Time-Travel & Forking: Because roster_state is saved on every row, you can query Turn 3 of a failed run, pull out the exact JSONB dictionary of mutated agent prompts, tweak one sentence, and resume the swarm from Turn 4.
• Mutation Auditing: You can query SELECT active_agent, payload->>'peer_name', payload->>'new_system_prompt' FROM swarm_ledger WHERE event_type = 'roster_mutation' to see which agent prompts your swarm rewrites most frequently.
• JSONL Export Anytime: If you ever want a flat JSONL file for another tool, a single psql one-liner exports any run cleanly:bash
psql -d swarm_experiments -t -A -c "SELECT row_to_json(t) FROM (SELECT * FROM swarm_ledger WHERE run_id = 'your-uuid' ORDER BY event_id) t;" > run.jsonl

what is the best way for these ephemeral sub-agents and self-modifying peer roster agents to communicate with each other after they are created?

The best communication architecture is a shared Postgres-backed Blackboard (structured state + event log) paired with lightweight, typed envelopes—rather than passing raw, full-history chat transcripts between agents.
When you pass full conversation histories from agent to agent, local models in LM Studio quickly suffer from context rot, prompt bloat, and KV-cache invalidation. Since you are already setting up a Postgres ledger, using that exact same database as a structured shared workspace (Blackboard) gives both patterns clean, auditable communication.

How Communication Should Work for Each Pattern


Dimension	Ephemeral Sub-Agents (Hierarchical)	Self-Modifying Peer Roster (Lateral)
Primary Channel	Typed Request/Response Envelopes. Parent passes a strict JSON brief; child returns a structured JSON artifact.	Shared Blackboard + Direct Mentions. Peers read/write named artifacts in Postgres and pass short pointers via handoff messages.
Context Visibility	Isolated. Sub-agents never see the parent's full history—only the scoped task and specific blackboard keys.	Scoped + Event Stream. Peers see the current blackboard state plus the last N structured handoff summaries.
Peer-to-Peer Talk	None (Hub-and-Spoke). If Worker A needs Worker B, it returns a "needs_input_from" status to the Architect.	Direct & Asynchronous. Agent A writes an artifact to the blackboard and calls handoff(target="agent_b", artifact_id=...).


1. The Shared Blackboard Pattern (Best for Peer Rosters)

Instead of agents stuffing massive drafts or code blocks into handoff(message=...), give every agent two simple Postgres-backed tools: write_artifact(key, content) and read_artifact(key).
• How they talk: planner writes its proposal to artifact:migration_plan_v1, then sends a tiny, 2-sentence handoff to critic: "I wrote migration_plan_v1 to the blackboard. Focus on locking risks."
• Why it works: The handoff message stays under 100 tokens, keeping turns fast on your Mac, while the actual work product lives cleanly versioned in Postgres (SELECT content FROM artifacts WHERE run_id = ... AND key = 'migration_plan_v1').

can both structured envelopes and peer mailboxes be accomplished using postgres (or is there a better way)?

Yes—PostgreSQL can handle both structured envelopes and peer mailboxes cleanly, and for a single-machine local experiment on your Mac, it is actually the best tool for the job.
You do not need Redis, RabbitMQ, or Kafka. Postgres gives you ACID transactions, JSONB schema flexibility, row-level locking (FOR UPDATE SKIP LOCKED), and LISTEN/NOTIFY for instant wake-ups—meaning your entire communication bus and immutable audit ledger live in one database with zero extra infrastructure.

1. How Postgres Handles Both Patterns Together

You can support both synchronous envelopes (for ephemeral workers) and asynchronous peer mailboxes (for concurrent or turn-based rosters) with two lightweight tables alongside your ledger:
sql

    author_agent TEXT NOT NULL,
    content      JSONB NOT NULL,
    created_at   TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    PRIMARY KEY (run_id, artifact_key, version)
);

-- 2. Peer Mailbox & Structured Envelope Queue
CREATE TABLE swarm_mailbox (
    msg_id       BIGSERIAL PRIMARY KEY,
    run_id       UUID NOT NULL,
    sender_agent TEXT NOT NULL,
    target_agent TEXT NOT NULL,
    msg_type     TEXT NOT NULL,        -- 'envelope_reply', 'handoff', 'critique', 'broadcast'
    envelope     JSONB NOT NULL,       -- Validated structured payload (status, summary, artifact_key)
    status       TEXT NOT NULL DEFAULT 'unread', -- 'unread', 'processing', 'processed'
    created_at   TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_mailbox_inbox ON swarm_mailbox (run_id, target_agent, status, msg_id);

2. Why Postgres Excels as an Agent Mailbox

• Safe Concurrent Workers (SKIP LOCKED): If you run 3 peers concurrently using Python asyncio, two workers will never accidentally grab the same task or overwrite each other if you fetch inbox messages with:sql
UPDATE swarm_mailbox
SET status = 'processing'
WHERE msg_id IN (
    SELECT msg_id FROM swarm_mailbox
    WHERE run_id = %s AND target_agent = %s AND status = 'unread'
    ORDER BY msg_id
    FOR UPDATE SKIP LOCKED
)
RETURNING msg_id, sender_agent, msg_type, envelope;
Use code with caution.
• Zero-Polling Wakeups (LISTEN / NOTIFY): Instead of burning CPU polling Postgres in a while loop, you can attach a tiny trigger on swarm_mailbox (NOTIFY swarm_wake, 'target_agent') so your Python orchestrator wakes up the exact millisecond a sub-agent drops an envelope into the table.
• Unified Replay: Because every blackboard artifact and mailbox envelope is timestamped in Postgres, a single SQL join reconstructs the exact state of every agent's inbox at any turn.

The only time Postgres isn't enough is if you scale to thousands of concurrent agents across distributed cloud nodes (where NATS, Kafka, or Temporal handles high-throughput event streaming better) or if agents need streaming token-by-token collaboration mid-generation (where WebSockets or Redis Streams fit better). For local multi-agent experiments on a Mac, Postgres is the sweet spot.

can peer agents decide to be ephemeral?

Yes—in fact, giving peer agents a retire_self (or spawn_ephemeral_peer) capability is one of the cleanest ways to keep a mutable swarm from getting bloated and noisy.
In a pure Postgres-backed roster, "ephemeral" simply means an agent has a lifecycle state (active vs. retired) or a ttl_turns (time-to-live) column in your swarm_roster table. Once a specialist finishes its narrow job and drops its structured envelope into swarm_mailbox or swarm_artifacts, it can either self-terminate or be created with a built-in expiration.

Three Ways a Peer Can Be Ephemeral

1. Self-Retirement (retire_self tool): A peer finishes its task (e.g., sql_migration_auditor reviews the schema), writes its findings to swarm_artifacts, sends a handoff envelope to planner, and calls retire_self(reason="Schema audit complete"). Your Python loop marks status = 'retired' in Postgres so it no longer appears in future roster snapshots or consumes context.
2. TTL-Bound Spawning (ttl_turns=1): When Peer A recruits Peer B via upsert_peer, Peer A can specify ttl_turns=1 (single-shot worker) vs. ttl_turns=None (persistent peer). After Peer B executes one turn and posts its reply envelope, the orchestrator automatically retires it.
3. Auto-Garbage Collection (Idle Eviction): Your orchestrator loop automatically retires any non-core peer that hasn't received a mailbox message in N turns, keeping the live roster lean.

How This Looks in Your Postgres Schema

Adding a lifecycle column to a swarm_roster table makes ephemeral peers first-class citizens alongside persistent peers, while preserving full auditability in your ledger:
sql
CREATE TABLE swarm_roster (
    run_id        UUID NOT NULL,
    agent_name    TEXT NOT NULL,
    system_prompt TEXT NOT NULL,
    lifecycle     TEXT NOT NULL DEFAULT 'persistent', -- 'persistent' | 'ephemeral'
    ttl_turns     INT,                                -- NULL = unlimited, 1 = single-shot
    turns_used    INT NOT NULL DEFAULT 0,
    status        TEXT NOT NULL DEFAULT 'active',     -- 'active' | 'retired'
    created_by    TEXT NOT NULL,
    updated_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    PRIMARY KEY (run_id, agent_name)
);

Why Combining Both Patterns Works So Well

• Zero Roster Bloat: Agents don't have to read through 10 dead-end specialists in their CURRENT LIVE ROSTER prompt block—they only query WHERE status = 'active'.
• Clean Handoff Hygiene: If an ephemeral agent retires itself without handing off to an active peer, your loop simply routes control back to created_by (the peer that spawned it) along with the retiree's final envelope.

class SwarmDB:
    def __init__(self, dsn: str):
        self.conn = psycopg.connect(dsn, row_factory=dict_row)
        with self.conn.cursor() as cur:
            cur.execute(SCHEMA_SQL)
        self.conn.commit()

    def get_active_roster(self, run_id: uuid.UUID) -> list[dict]:
        with self.conn.cursor() as cur:
            cur.execute(
                "SELECT * FROM swarm_roster WHERE run_id = %s AND status = 'active' ORDER BY agent_name",
                (run_id,),
            )
            return cur.fetchall()

    def get_agent(self, run_id: uuid.UUID, agent_name: str) -> dict | None:
        with self.conn.cursor() as cur:
            cur.execute(
                "SELECT * FROM swarm_roster WHERE run_id = %s AND agent_name = %s",
                (run_id, agent_name),
            )
            return cur.fetchone()

    def upsert_agent(
        self,
        run_id: uuid.UUID,
        agent_name: str,
        system_prompt: str,
        lifecycle: str,
        ttl_turns: int | None,
        created_by: str,
    ):
        with self.conn.cursor() as cur:
            cur.execute(
                """
                INSERT INTO swarm_roster (run_id, agent_name, system_prompt, lifecycle, ttl_turns, status, created_by)
                VALUES (%s, %s, %s, %s, %s, 'active', %s)
                ON CONFLICT (run_id, agent_name) DO UPDATE SET
                    system_prompt = EXCLUDED.system_prompt,
                    lifecycle = EXCLUDED.lifecycle,
                    ttl_turns = EXCLUDED.ttl_turns,
                    status = 'active',
                    updated_at = NOW()
                """,
                (run_id, agent_name, system_prompt, lifecycle, ttl_turns, created_by),
            )
        self.conn.commit()

    def increment_turn_and_check_ttl(self, run_id: uuid.UUID, agent_name: str) -> bool:
        """Increments turns_used; returns True if agent hit its TTL and was auto-retired."""
        with self.conn.cursor() as cur:
            cur.execute(
                """
                UPDATE swarm_roster
                SET turns_used = turns_used + 1, updated_at = NOW()
                WHERE run_id = %s AND agent_name = %s
                RETURNING lifecycle, ttl_turns, turns_used
                """,
                (run_id, agent_name),
            )
            row = cur.fetchone()
            if row and row["ttl_turns"] is not None and row["turns_used"] >= row["ttl_turns"]:
                cur.execute(
                    "UPDATE swarm_roster SET status = 'retired', updated_at = NOW() WHERE run_id = %s AND agent_name = %s",
                    (run_id, agent_name),
                )
                self.conn.commit()
                return True
        self.conn.commit()
        return False

    def retire_agent(self, run_id: uuid.UUID, agent_name: str):
        with self.conn.cursor() as cur:
            cur.execute(
                "UPDATE swarm_roster SET status = 'retired', updated_at = NOW() WHERE run_id = %s AND agent_name = %s",
                (run_id, agent_name),
            )
        self.conn.commit()

    def write_artifact(self, run_id: uuid.UUID, key: str, author: str, content: str) -> int:
        with self.conn.cursor() as cur:
            cur.execute(
                "SELECT COALESCE(MAX(version), 0) + 1 AS next_v FROM swarm_artifacts WHERE run_id = %s AND artifact_key = %s",
                (run_id, key),
            )
            next_v = cur.fetchone()["next_v"]
            cur.execute(
                """
                INSERT INTO swarm_artifacts (run_id, artifact_key, version, author_agent, content)
                VALUES (%s, %s, %s, %s, %s)
                """,
                (run_id, key, next_v, author, content),
            )
        self.conn.commit()
        return next_v

def pop_unread_mail(self, run_id: uuid.UUID, target: str) -> list[dict]:
with self.conn.cursor() as cur:

"""
UPDATE swarm_mailbox SET status = 'processed'
WHERE msg_id IN (
SELECT msg_id FROM swarm_mailbox
WHERE run_id = %s AND target_agent = %s AND status = 'unread'
ORDER BY msg_id FOR UPDATE SKIP LOCKED
)
RETURNING msg_id, sender_agent, msg_type, envelope, created_at
""",
(run_id, target),
)
rows = cur.fetchall()
self.conn.commit()
return rows
def log_event(
self,
run_id: uuid.UUID,
turn: int,
event_type: str,
active_agent: str,
payload: dict,
target_agent: str | None = None,
):

(run_id, turn, event_type, active_agent, target_agent, json.dumps(payload), json.dumps(roster_snapshot)),

f"BLACKBOARD ARTIFACTS: {artifact_text}\n\n"
"WORKFLOW RULES:\n"
"1. Store heavy drafts/code in write_artifact and reference the artifact_key in send_handoff.\n"
"2. If you are 'ephemeral', call retire_self after writing your artifact/handoff.\n"

)

]

while True:

if agent_row["lifecycle"] == "ephemeral":

sender=active_agent,
target=fallback,

)

break
else:

print(f"\n=== SWARM CONCLUDED BY {active_agent} ===\n{msg.content}")
return run_id
for call in msg.tool_calls:
fn = call.function.name
args = json.loads(call.function.arguments)
tool_result = ""
if fn == "write_artifact":

print(f"  [ROSTER] {active_agent} upserted '{peer}' ({lifecycle}, ttl={ttl})")
tool_result = f"Peer '{peer}' is now active in the roster."
elif fn == "send_handoff":
target = args["target_peer"].lower().strip()
if not db.get_agent(run_id, target):
tool_result = f"Error: Peer '{target}' does not exist. Create them with upsert_peer first."
else:
envelope = {
"summary": args["summary"],
"artifact_key": args.get("artifact_key"),
"instructions": args["instructions"],
}
db.send_mail(run_id, active_agent, target, "handoff", envelope)
db.log_event(run_id, turn, "handoff", active_agent, envelope, target_agent=target)
print(f"  [MAILBOX] {active_agent} -> {target}: {args['summary']}")
next_agent = target
tool_result = f"Envelope queued for '{target}'."
elif fn == "retire_self":
db.retire_agent(run_id, active_agent)
fallback = next_agent or (agent_row["created_by"] if agent_row["created_by"] != "system" else "planner")
envelope = {"summary": args["reason"], "artifact_key": args.get("artifact_key"), "instructions": "Review retired specialist output."}
db.send_mail(run_id, active_agent, fallback, "retire_notice", envelope)
db.log_event(run_id, turn, "retire_self", active_agent, envelope, target_agent=fallback)
print(f"  [RETIRED] {active_agent} retired -> returning control to {fallback}")
next_agent = fallback
tool_result = "You are now retired."
messages.append({"role": "tool", "tool_call_id": call.id, "content": tool_result})
# Exit inner tool loop once a handoff or self-retirement has occurred
if next_agent:
break
# Enforce TTL auto-retirement after the turn completes
if db.increment_turn_and_check_ttl(run_id, active_agent):
print(f"  [TTL EXPIRED] '{active_agent}' reached its turn limit and was auto-retired.")
active_agent = next_agent or active_agent
print("\n=== MAX TURNS REACHED ===")
return run_id

Yes, that is completely viable—and it is actually a cleaner, superior architecture.
Unifying everything into a single peer model where ttl_turns dictates lifespan (NULL = persistent architect/coordinator, 1 or N = ephemeral worker) eliminates redundant concepts, schema columns, and branching logic in your orchestrator.

Why a Unified TTL-Bound Peer Model Is Better

• Simpler Postgres Schema: You can drop the lifecycle column from swarm_roster entirely. ttl_turns IS NULL means persistent, ttl_turns IS NOT NULL means bounded/ephemeral, and any agent can still call retire_self early if it finishes ahead of its TTL.
• Uniform Communication: Every agent—whether it lives for 1 turn or 50 turns—uses the exact same swarm_mailbox and swarm_artifacts blackboard. You don't need separate synchronous function-return logic for ephemeral sub-agents versus mailbox logic for peers.
• Natural Guardrail Against Runaway Loops: Setting a default ttl_turns=3 on newly recruited specialists guarantees that even if a local model forgets to call retire_self or gets confused, the orchestrator automatically garbage-collects it and routes control back to created_by.

TTL expiry handles normal turn-based execution, so you do not need a separate background garbage collector—with one small exception for "orphaned" peers.
Because your orchestrator runs a single sequential loop and increments turns_used every time an agent takes a turn, turns_used >= ttl_turns deterministically retires any worker that actually gets invoked.

There is only one scenario where ttl_turns alone will not retire an ephemeral peer: if Peer A creates Peer B (turns_used = 0), but never hands off to Peer B (for example, Peer A changes its mind and hands off to Peer C instead).

Isn't the ledger just postgres driven function calling and not really AI agents

The ledger itself isn't the agent—it is the deterministic flight recorder, while the LLM sitting inside the loop making non-deterministic routing, tool, and mutation decisions is the agent.
That intuition hits on an important truth about modern AI engineering: once you strip away the marketing buzzwords, every AI agent framework is just a deterministic state machine (a while loop + database/memory + function-calling dispatch) wrapped around an LLM's next-token predictions.

Where the "Agent" Stops and the "Plumbing" Starts

• What Postgres + Python are doing (The Environment & Physics): Postgres is just the durable memory, mailbox, and audit log (swarm_ledger), and Python is the physics engine enforcing boundaries (TTL turn limits, tool allowlists, and jq output contracts). Neither Python nor Postgres decides what to do next.
• What makes it an AI Agent (The Policy): At each step, the LLM inspects the inbox, roster, and blackboard, and autonomously decides:
	• Which tool to call and with what arguments (e.g., choosing normalize_keys=true after reading the schema).
	• Whether to rewrite a peer's system prompt or recruit a new TTL-bound specialist via upsert_peer.
	• When the work is done versus when to hand off to a critic.

Why Your Benchmark Tests Look So Deterministic

Notice that in revenue-analysis.yaml and employee-audit.yaml, you intentionally constrained the swarm with narrow allowed_tools per phase and strict system prompts so that your jq assertions pass reliably on a local Mac model.
When you tighten constraints that heavily, a multi-agent swarm starts behaving very similarly to a state machine with LLM-powered transitions—which is actually what you want for reliable, testable software. If you loosen those constraints on an open-ended research or debugging problem (giving a Lead Architect full upsert_peer freedom with ttl_turns=None), the exact same Postgres loop exhibits much more emergent, unpredictable agent behavior.
